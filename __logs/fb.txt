### 1. **File Structure**

### 2. **Core Features**

#### Animation System
- **Framer Motion** cho spring animations khi expand/collapse
- Variants cho các states: `compact`, `expanded`, `hidden`
- Spring config: `{ type: "spring", stiffness: 300, damping: 30 }`

#### Queue Management
- FIFO queue để handle multiple notifications
- Auto-dismiss sau timeout (configurable, default: 5000ms)
- Show next item khi current item dismissed

#### State Management
```typescript

  duration?: number;

```

import React, { useState, useEffect, useCallback } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { Bell, Check, X, AlertCircle, Music, Phone, Mail } from 'lucide-react';

// ========================
// TYPES & INTERFACES
// ========================

interface NotificationItem {
  id: string;
  title: string;
  description?: string;
  icon?: React.ReactNode;
  action?: {
    label: string;
    onClick: () => void;
  };
  duration?: number;
  variant?: 'default' | 'success' | 'error' | 'info';
}

interface DynamicIslandProps {
  position?: 'top' | 'bottom';
  className?: string;
}

// ========================
// CUSTOM HOOK - useNotificationQueue
// ========================

const useNotificationQueue = () => {
  const [queue, setQueue] = useState<NotificationItem[]>([]);
  const [current, setCurrent] = useState<NotificationItem | null>(null);
  const [isExpanded, setIsExpanded] = useState(false);

  const addNotification = useCallback((item: Omit<NotificationItem, 'id'>) => {
    const notification: NotificationItem = {
      ...item,
      id: `notif-${Date.now()}-${Math.random()}`,
      duration: item.duration || 5000,
    };
    
    setQueue(prev => [...prev, notification]);
  }, []);

  const dismissCurrent = useCallback(() => {
    setIsExpanded(false);
    setTimeout(() => setCurrent(null), 300);
  }, []);

  // Process queue
  useEffect(() => {
    if (!current && queue.length > 0) {
      const [next, ...rest] = queue;
      setCurrent(next);
      setQueue(rest);
      
      // Auto expand
      setTimeout(() => setIsExpanded(true), 100);
    }
  }, [current, queue]);

  // Auto dismiss
  useEffect(() => {
    if (current && isExpanded) {
      const timer = setTimeout(() => {
        dismissCurrent();
      }, current.duration);

      return () => clearTimeout(timer);
    }
  }, [current, isExpanded, dismissCurrent]);

  return {
    current,
    isExpanded,
    queueLength: queue.length,
    addNotification,
    dismissCurrent,
  };
};

// ========================
// VARIANT STYLES
// ========================

const variantStyles = {
  default: 'bg-gradient-to-r from-gray-900 to-gray-800 border-gray-700',
  success: 'bg-gradient-to-r from-green-900 to-green-800 border-green-700',
  error: 'bg-gradient-to-r from-red-900 to-red-800 border-red-700',
  info: 'bg-gradient-to-r from-blue-900 to-blue-800 border-blue-700',
};

// ========================
// MAIN COMPONENT
// ========================

const DynamicIsland: React.FC<DynamicIslandProps> = ({ 
  position = 'top',
  className = '' 
}) => {
  const { current, isExpanded, queueLength, dismissCurrent } = useNotificationQueue();

  if (!current) return null;

  const positionClass = position === 'top' ? 'top-6' : 'bottom-6';
  const variantClass = variantStyles[current.variant || 'default'];

  return (
    <motion.div
      className={`fixed ${positionClass} left-1/2 -translate-x-1/2 z-50 ${className}`}
      initial={{ opacity: 0, scale: 0.8, y: position === 'top' ? -20 : 20 }}
      animate={{ opacity: 1, scale: 1, y: 0 }}
      exit={{ opacity: 0, scale: 0.8, y: position === 'top' ? -20 : 20 }}
      transition={{ type: 'spring', stiffness: 300, damping: 25 }}
    >
      <motion.div
        layout
        className={`${variantClass} backdrop-blur-xl border shadow-2xl overflow-hidden`}
        animate={{
          width: isExpanded ? '380px' : '140px',
          height: isExpanded ? 'auto' : '44px',
          borderRadius: isExpanded ? '24px' : '22px',
        }}
        transition={{ type: 'spring', stiffness: 300, damping: 30 }}
      >
        <AnimatePresence mode="wait">
          {!isExpanded ? (
            <CompactView key="compact" icon={current.icon} queueLength={queueLength} />
          ) : (
            <ExpandedView 
              key="expanded" 
              item={current}
              onDismiss={dismissCurrent}
            />
          )}
        </AnimatePresence>
      </motion.div>
    </motion.div>
  );
};

// ========================
// COMPACT VIEW
// ========================

const CompactView: React.FC<{ icon?: React.ReactNode; queueLength: number }> = ({ 
  icon, 
  queueLength 
}) => (
  <motion.div
    initial={{ opacity: 0 }}
    animate={{ opacity: 1 }}
    exit={{ opacity: 0 }}
    className="flex items-center justify-center h-[44px] px-4"
  >
    <div className="flex items-center gap-2">
      {icon && <div className="text-white/90 scale-90">{icon}</div>}
      <div className="w-1.5 h-1.5 rounded-full bg-white/60 animate-pulse" />
      {queueLength > 0 && (
        <span className="text-xs text-white/60 font-medium">+{queueLength}</span>
      )}
    </div>
  </motion.div>
);

// ========================
// EXPANDED VIEW
// ========================

const ExpandedView: React.FC<{ 
  item: NotificationItem;
  onDismiss: () => void;
}> = ({ item, onDismiss }) => (
  <motion.div
    initial={{ opacity: 0 }}
    animate={{ opacity: 1 }}
    exit={{ opacity: 0 }}
    className="p-4"
  >
    <div className="flex items-start gap-3">
      {item.icon && (
        <div className="text-white/90 mt-0.5 flex-shrink-0">
          {item.icon}
        </div>
      )}
      
      <div className="flex-1 min-w-0">
        <h3 className="text-white font-semibold text-sm mb-1">
          {item.title}
        </h3>
        {item.description && (
          <p className="text-white/70 text-xs leading-relaxed">
            {item.description}
          </p>
        )}
        
        {item.action && (
          <button
            onClick={() => {
              item.action?.onClick();
              onDismiss();
            }}
            className="mt-3 px-4 py-1.5 bg-white/10 hover:bg-white/20 text-white text-xs font-medium rounded-full transition-colors"
          >
            {item.action.label}
          </button>
        )}
      </div>
      
      <button
        onClick={onDismiss}
        className="text-white/50 hover:text-white/90 transition-colors flex-shrink-0"
        aria-label="Dismiss"
      >
        <X size={16} />
      </button>
    </div>
  </motion.div>
);

## 📦 **Architecture Highlights**

### 1. **Separation of Concerns**
- **Types**: Tất cả interfaces ở đầu file
- **Custom Hook**: `useNotificationQueue` quản lý toàn bộ logic
- **Sub-components**: `CompactView` và `ExpandedView` tách biệt
- **Main Component**: Chỉ lo presentation layer

### 2. **Key Features**
✅ **Queue Management** - FIFO, xử lý multiple notifications  
✅ **Smooth Animations** - Spring physics với Framer Motion  
✅ **Auto-dismiss** - Configurable duration (default 5s)  
✅ **Variants** - Success, Error, Info, Default với colors riêng  
✅ **Actions** - Support buttons với callbacks  
✅ **TypeScript** - Fully typed, type-safe  

### 3. **Demo Includes**
- 5 notification types khác nhau
- Interactive test buttons
- Features showcase
- Professional UI/UX
